    async function fetchWithTimeout(url, timeoutMs = 4500) {

    function loadSelectedWeatherPlaces() {
      const raw = localStorage.getItem(WEATHER_REGIONS_KEY);

    function toggleWeatherPlace(placeName) {
      const idx = selectedWeatherPlaces.indexOf(placeName);
      if (idx !== -1) {
        if (selectedWeatherPlaces.length <= 1) {
          alert('請至少保留 1 個天氣顯示地區！');
          return;
        }
        selectedWeatherPlaces.splice(idx, 1);
      } else {
        selectedWeatherPlaces.push(placeName);
      }
      saveSelectedWeatherPlaces();
      renderWeatherRegionsBar();
      renderWeatherStationsModalGrid();
    }

    function openWeatherModal() {
      renderWeatherStationsModalGrid();

    }

    function renderWeatherStationsModalGrid() {
      const grid = document.getElementById('weather-stations-grid');
      const knownPlaces = new Set(HKO_WEATHER_STATIONS.map(s => s.place));
      const allStations = [...HKO_WEATHER_STATIONS];
      latestHkoTempMap.forEach((val, key) => {
        if (!knownPlaces.has(key)) allStations.push({ place: key, district: '香港天文台' });
      });

      grid.innerHTML = allStations.map(st => {
        const isSelected = selectedWeatherPlaces.includes(st.place);
        const tempVal = latestHkoTempMap.has(st.place) ? `${latestHkoTempMap.get(st.place)}°C` : '--°C';
        return `
          <button type="button" onclick="toggleWeatherPlace('${st.place}')"
            class="flap-row p-2.5 flex items-center justify-between text-left transition ${isSelected ? 'ring-2 ring-sky-500' : 'opacity-75 hover:opacity-100'}">
            <div class="min-w-0">
              <div class="text-xs font-black text-pri truncate">${st.place}</div>
              <div class="text-[10px] font-bold text-mut truncate">${st.district}</div>
            </div>
            <div class="flex items-center gap-1.5 shrink-0">
              <span class="flap-tile px-1.5 py-0.5 text-xs font-extrabold mono-num ${isSelected ? 'flap-amber' : ''}">${tempVal}</span>
              <span class="text-xs font-black ${isSelected ? 'text-sky-400' : 'text-mut'}">${isSelected ? '✓' : '＋'}</span>
            </div>
          </button>
        `;
      }).join('');
    }

    function renderWeatherRegionsBar() {
      const container = document.getElementById('weather-regions-container');

        return `
          <div class="flap-row px-2.5 py-1 flex items-center gap-2">
            <div>
              <div class="text-[10px] font-black text-mut leading-none mb-0.5">${place}</div>

            </div>
            ${selectedWeatherPlaces.length > 1 ? `

    async function fetchWeather() {
      try {
        const [rhr, warn] = await Promise.all([
          fetchJsonWithTimeout('https://data.weather.gov.hk/weatherAPI/opendata/weather.php?dataType=rhrread&lang=tc', 4500),
          fetchJsonWithTimeout('https://data.weather.gov.hk/weatherAPI/opendata/weather.php?dataType=warnsum&lang=tc', 4500)
        ]);

        const tempArr = rhr?.temperature?.data || [];
        latestHkoTempMap.clear();
        tempArr.forEach(t => {
          if (t && t.place && t.value !== undefined) latestHkoTempMap.set(t.place, t.value);
        });

        const humObj = rhr?.humidity?.data?.[0];
        if (humObj && humObj.value !== undefined) latestHkoHumidity = humObj.value;

        const iconCode = rhr?.icon?.[0];
        if (iconCode) {
          const iconImg = document.getElementById('weather-icon');
          iconImg.src = `https://www.hko.gov.hk/images/HKOWxIconOutline/pic${iconCode}.png`;
          iconImg.classList.remove('hidden');
          document.getElementById('weather-icon-fallback').classList.add('hidden');
        }

        const warnContainer = document.getElementById('weather-warnings');
        warnContainer.innerHTML = '';
        if (warn && typeof warn === 'object') {
          Object.values(warn).forEach(w => {
            if (w && w.name) {
              const badge = document.createElement('span');
              badge.className = 'flap-tile px-1.5 py-0.5 text-[10px] font-black flap-amber truncate';
              badge.textContent = w.name;
              warnContainer.appendChild(badge);
            }
          });
        }
        renderWeatherRegionsBar();
      } catch (err) {
        renderWeatherRegionsBar();
      }
    }

    async function fetchTdTrafficNews() {
      const xmlUrl = 'https://resource.data.one.gov.hk/td/tc/specialtrafficnews.xml';
      let xmlText = '';

      try {
        const res = await fetchWithTimeout(xmlUrl, 3500);
        if (res.ok) xmlText = await res.text();
      } catch (e) {
        try {
          const proxyRes = await fetchWithTimeout(`https://api.allorigins.win/raw?url=${encodeURIComponent(xmlUrl)}`, 3500);
          if (proxyRes.ok) xmlText = await proxyRes.text();
        } catch (e2) {}
      }

      if (xmlText) {
        const parser = new DOMParser();
        const xmlDoc = parser.parseFromString(xmlText, 'text/xml');
        const msgNodes = Array.from(xmlDoc.querySelectorAll('message'));
        latestTdAllMessages = msgNodes.map(node => {
          const tc = node.querySelector('ChinText')?.textContent || node.textContent || '';
          return tc.replace(/\s+/g, ' ').trim();
        }).filter(Boolean);
      }

      await fetchRequiredMtrLinesStatus();
      evaluateAndRenderAllTrafficMonitors();
    }

    async function fetchRequiredMtrLinesStatus() {
      const mtrItems = trafficMonitors.filter(m => m.type === 'mtr');
      const uniqueLines = [...new Set(mtrItems.map(m => m.mtrLine || 'EAL'))];

      await Promise.all(uniqueLines.map(async (lineCode) => {
        const meta = MTR_LINES_META.find(l => l.code === lineCode) || MTR_LINES_META[0];
        try {
          const json = await fetchJsonWithTimeout(`https://rt.data.gov.hk/v1/transport/mtr/getSchedule.php?line=${lineCode}&sta=${meta.sampleSta}&lang=TC`, 4000);
          const tdAlert = latestTdAllMessages.find(msg => meta.keywords.some(k => msg.includes(k)));

          if (json.status === 0) {
            const msg = json.message || tdAlert || '列車服務受阻';
            const isSevere = msg.includes('暫停') || msg.includes('嚴重') || msg.includes('事故');
            mtrLiveStatusCache[lineCode] = { level: isSevere ? 'red' : 'yellow', badge: isSevere ? '嚴重事故' : '異常', sub: msg.slice(0, 26) };
          } else if (json.isdelay === 'Y' || tdAlert) {
            mtrLiveStatusCache[lineCode] = { level: 'yellow', badge: '延誤', sub: (tdAlert || '港鐵回報班次延誤').slice(0, 26) };
          } else {
            const downTrains = json?.data?.[`${lineCode}-${meta.sampleSta}`]?.DOWN || [];
            const nextMin = downTrains[0]?.ttnt;
            mtrLiveStatusCache[lineCode] = {
              level: 'green',
              badge: '正常',
              sub: nextMin ? `全線服務正常（下班約 ${nextMin} 分鐘）` : '全線列車服務正常'
            };
          }
        } catch (e) {
          mtrLiveStatusCache[lineCode] = { level: 'green', badge: '正常', sub: '列車服務正常運作' };
        }
      }));
    }

    function evaluateSingleMonitor(item) {
      if (item.type === 'mtr') {
        return mtrLiveStatusCache[item.mtrLine || 'EAL'] || { level: 'green', badge: '正常', sub: '列車服務正常' };
      }

      const keywords = item.keywords || [];
      const matchedNews = latestTdAllMessages.filter(msg => keywords.some(k => k && msg.includes(k)));

      if (matchedNews.length > 0) {
        const severeWords = ['全線封閉', '嚴重', '封路', '交通意外', '暫停', '單線雙程', '擠塞'];
        const isSevere = matchedNews.some(m => severeWords.some(w => m.includes(w)));
        return {
          level: isSevere ? 'red' : 'yellow',
          badge: isSevere ? '嚴重事故' : '異常',
          sub: matchedNews[0].slice(0, 26) + '...'
        };
      }

      if (item.use64kLiveEtaDiff && latestCorridorMinutes !== null) {
        if (latestCorridorMinutes >= 18) return { level: 'red', badge: '嚴重擠塞', sub: `實時車程約 ${latestCorridorMinutes} 分鐘` };
        if (latestCorridorMinutes >= 13) return { level: 'yellow', badge: '車多緩慢', sub: `實時車程約 ${latestCorridorMinutes} 分鐘` };
        return { level: 'green', badge: '正常', sub: `暢順・車程約 ${latestCorridorMinutes} 分鐘` };
      }

      const now = new Date();
      const day = now.getDay();
      const totalMin = now.getHours() * 60 + now.getMinutes();
      const isPeak = (day >= 1 && day <= 5) && ((totalMin >= 465 && totalMin <= 540) || (totalMin >= 1065 && totalMin <= 1155));

      if (isPeak && (item.type === 'tunnel' || item.type === 'corridor')) {
        return { level: 'yellow', badge: '繁忙', sub: `${item.subLabel || '繁忙時段車多'}` };
      }

      return {
        level: 'green',
        badge: '正常',
        sub: item.subLabel ? `${item.subLabel}・交通暢順` : '道路交通暢順'
      };
    }

    function evaluateAndRenderAllTrafficMonitors() {
      const grid = document.getElementById('traffic-monitors-grid');
      if (!grid) return;

      const allRelevantIncidents = [];

      grid.innerHTML = trafficMonitors.map((item, idx) => {

        const dotClass = res.level === 'red' ? 'dot-red' : res.level === 'yellow' ? 'dot-yellow' : 'dot-green';

    async function saveTrafficMonitorItem() {
      let newItem = null;

      if (currentTrafficModalCat === 'district') {
        const code = document.getElementById('tm-dist-select').value;
        const sub = document.getElementById('tm-subarea-select').value;
        const dist = HK_18_DISTRICTS.find(d => d.code === code);
        if (!dist) return;

        const cleanSubs = dist.subAreas.map(s => s.replace(/（.*）/, ''));
        const isAll = sub === 'ALL';
        const title = isAll ? `${dist.name}交通` : `${dist.name}（${sub}）`;
        const subLabel = isAll ? dist.subAreas.slice(0, 3).join(' / ') : `${sub}及周邊幹道`;
        const keywords = isAll ? [dist.name.replace('區', ''), ...cleanSubs, ...dist.roads] : [sub.replace(/（.*）/, ''), ...dist.roads.slice(0, 4)];

        newItem = { id: 'tm_' + Date.now(), type: 'district', title, districtCode: code, subArea: sub, subLabel, keywords };
      } else if (currentTrafficModalCat === 'corridor') {
        const codeA = document.getElementById('tm-corr-dist-a').value;
        const subA  = document.getElementById('tm-corr-sub-a').value;
        const codeB = document.getElementById('tm-corr-dist-b').value;
        const subB  = document.getElementById('tm-corr-sub-b').value;

        const dA = HK_18_DISTRICTS.find(d => d.code === codeA);
        const dB = HK_18_DISTRICTS.find(d => d.code === codeB);
        const labelA = subA === 'ALL' ? dA.name : subA.replace(/（.*）/, '');
        const labelB = subB === 'ALL' ? dB.name : subB.replace(/（.*）/, '');

        const keywords = buildConnectingCorridorKeywords(codeA, subA, codeB, subB);
        const isChungUkTaiWo = (labelA.includes('林村') || labelA.includes('鍾屋村')) && labelB.includes('太和');

        newItem = {
          id: 'tm_' + Date.now(),
          type: 'corridor',
          title: `${labelA} ➔ ${labelB}`,
          subLabel: keywords.slice(0, 2).join(' / '),
          keywords,
          use64kLiveEtaDiff: isChungUkTaiWo
        };
      } else if (currentTrafficModalCat === 'road') {
        const roadName = document.getElementById('tm-road-custom').value.trim();
        if (!roadName) return;
        newItem = { id: 'tm_' + Date.now(), type: 'road', title: roadName, subLabel: '指定道路實時監測', keywords: [roadName] };
      } else if (currentTrafficModalCat === 'mtr') {
        const lineCode = document.getElementById('tm-mtr-select').value;
        const meta = MTR_LINES_META.find(l => l.code === lineCode);
        if (!meta) return;
        newItem = { id: 'tm_' + Date.now(), type: 'mtr', title: meta.name, mtrLine: meta.code, subLabel: '港鐵實時班次監測', keywords: meta.keywords };
      } else if (currentTrafficModalCat === 'tunnel') {
        const tIdx = Number(document.getElementById('tm-tunnel-select').value);
        const tObj = HK_TUNNELS_AND_BRIDGES[tIdx];
        if (!tObj) return;
        newItem = { id: 'tm_' + Date.now(), type: 'tunnel', title: tObj.name, subLabel: '隧道/橋梁管制區', keywords: tObj.keywords };
      }

      if (!newItem) return;

      if (editingTrafficIdx >= 0) trafficMonitors[editingTrafficIdx] = newItem;
      else trafficMonitors.push(newItem);

      saveTrafficMonitors();
      closeTrafficModal();
      await fetchRequiredMtrLinesStatus();
      evaluateAndRenderAllTrafficMonitors();
    }

    function updateJrApproachBoard(slotIdx, card, etas, exactStopPositionPercent = null) {

        statusTextEl.className = 'text-slate-400 font-bold truncate';
        return;
      }

      const now = Date.now();
      const vehicleIcon = card.operator === 'GMB' ? '🚐' : card.operator === 'MTR' ? '🚆' : '🚌';

      const computeTrackPercentFromSec = (diffSec) => {
        const secPerStop = card.operator === 'MTR' ? 130 : 105;
        const maxTrackSec = secPerStop * 4;
        if (diffSec <= 20) return 100;
        if (diffSec >= maxTrackSec) return 0;
        return Math.max(0, Math.min(100, Math.round((1 - (diffSec / maxTrackSec)) * 100)));
      };

      const actorsHtml = [];

      etas.slice(0, 2).forEach((item, vIdx) => {
        const etaMs = new Date(item.timestamp).getTime();
        const diffSec = Math.max(0, Math.floor((etaMs - now) / 1000));
        const diffMin = Math.max(0, Math.floor(diffSec / 60));

        let percent = (vIdx === 0 && exactStopPositionPercent !== null)
          ? exactStopPositionPercent
          : computeTrackPercentFromSec(diffSec);

        if (vIdx === 1 && diffSec > 480) return;
        if (vIdx === 1 && actorsHtml.length > 0) percent = Math.max(0, percent - 12);

        const nearestNodeIdx = Math.min(4, Math.max(0, Math.round(percent / 25)));
        const nodeEl = document.getElementById(`jr-node-${slotIdx}-${nearestNodeIdx}`);
        if (nodeEl) {
          if (nearestNodeIdx === 4 && diffSec <= 60) nodeEl.classList.add('is-arriving');
          else nodeEl.classList.add('is-lit');
        }

        if (vIdx === 0) {
          const stopNames = card.fiveStops || ['前4站', '前3站', '前2站', '上1站', card.stopName];
          if (diffSec <= 60) {
            statusTextEl.textContent = `🟢 頭班車即將進站 (${shortStopName(stopNames[4])})`;
            statusTextEl.className = 'text-emerald-400 font-black truncate animate-pulse';
          } else if (diffSec > 450) {
            statusTextEl.textContent = `預計 ${diffMin} 分鐘後到達（距本站 4 站以上）`;
            statusTextEl.className = 'text-amber-300 font-bold truncate';
          } else {
            const fromStopIdx = Math.min(3, Math.floor(percent / 25));
            const toStopIdx = fromStopIdx + 1;
            const stopsAway = 4 - fromStopIdx;
            statusTextEl.textContent = `距本站 ${stopsAway} 站｜正由【${shortStopName(stopNames[fromStopIdx])}】駛往【${shortStopName(stopNames[toStopIdx])}】`;
            statusTextEl.className = 'text-amber-300 font-black truncate';
          }
        }

        const badgeLabel = diffSec <= 60 ? '進站中' : `${diffMin}分`;
        const pillClass = vIdx === 1 ? 'jr-bus-pill is-second' : (diffSec <= 60 ? 'jr-bus-pill is-arriving' : 'jr-bus-pill');

        actorsHtml.push(`
          <div class="jr-bus-actor" style="left: ${percent}%;">
            <div class="${pillClass}">
              <span>${vehicleIcon}</span>
              <span class="mono-num">${badgeLabel}</span>
            </div>
          </div>
        `);
      });

      vehiclesContainer.innerHTML = actorsHtml.join('');
    }

    let sunriseMinutes = 6 * 60 + 25;
    let sunsetMinutes  = 17 * 60 + 48;
    let sunriseStr = '06:25';
    let sunsetStr  = '17:48';
    const THEME_MODE_KEY = 'dashboard_theme_mode_v2';
    let themeMode = localStorage.getItem(THEME_MODE_KEY) || 'auto';

    function computeHkSunTimes(dateObj) {
      const start = new Date(dateObj.getFullYear(), 0, 0);
      const dayOfYear = Math.floor((dateObj - start) / 86400000);
      const lat = 22.445 * Math.PI / 180;
      const lng = 114.165;
      const gamma = (2 * Math.PI / 365) * (dayOfYear - 1);

      const eqTime = 229.18 * (0.000075 + 0.001868 * Math.cos(gamma) - 0.032077 * Math.sin(gamma)
                    - 0.014615 * Math.cos(2 * gamma) - 0.040849 * Math.sin(2 * gamma));
      const decl = 0.006918 - 0.399912 * Math.cos(gamma) + 0.070257 * Math.sin(gamma)
                  - 0.006758 * Math.cos(2 * gamma) + 0.000907 * Math.sin(2 * gamma)
                  - 0.002697 * Math.cos(3 * gamma) + 0.00148 * Math.sin(3 * gamma);

      const zenith = 90.833 * Math.PI / 180;
      const cosHA = (Math.cos(zenith) / (Math.cos(lat) * Math.cos(decl))) - Math.tan(lat) * Math.tan(decl);
      const ha = Math.acos(Math.min(1, Math.max(-1, cosHA))) * 180 / Math.PI;

      sunriseMinutes = Math.round(720 - 4 * (lng + ha) - eqTime + 480);
      sunsetMinutes  = Math.round(720 - 4 * (lng - ha) - eqTime + 480);

      const fmt = (m) => `${String(Math.floor(m / 60)).padStart(2, '0')}:${String(m % 60).padStart(2, '0')}`;
      sunriseStr = fmt(sunriseMinutes);
      sunsetStr  = fmt(sunsetMinutes);
    }

    async function fetchHkoSunTimes() {
      const now = new Date();
      computeHkSunTimes(now);
      try {
        const y = now.getFullYear();
        const m = now.getMonth() + 1;
        const d = now.getDate();
        const json = await fetchJsonWithTimeout(`https://data.weather.gov.hk/weatherAPI/opendata/opendata.php?dataType=SRS&rformat=json&year=${y}&month=${m}&day=${d}`, 4000);
        const row = json?.data?.[0];
        if (row && row[1] && row[3]) {
          sunriseStr = row[1];
          sunsetStr  = row[3];
          const [rH, rM] = sunriseStr.split(':').map(Number);
          const [sH, sM] = sunsetStr.split(':').map(Number);
          if (!isNaN(rH) && !isNaN(sH)) {
            sunriseMinutes = rH * 60 + rM;
            sunsetMinutes  = sH * 60 + sM;
          }
        }
      } catch (e) {}
      applyDayNightTheme(new Date());
    }

    function applyDayNightTheme(now) {
      const nowMin = now.getHours() * 60 + now.getMinutes();
      const isBeforeSunsetDaytime = (nowMin >= sunriseMinutes && nowMin < sunsetMinutes);

      let useLight = false;
      if (themeMode === 'light') useLight = true;
      else if (themeMode === 'dark') useLight = false;
      else useLight = isBeforeSunsetDaytime;

      document.body.classList.toggle('theme-light', useLight);
      document.body.classList.toggle('theme-dark', !useLight);

      document.getElementById('sun-theme-badge').textContent = `🌅 ${sunriseStr} 🌇 ${sunsetStr}`;
      const btnMap = { auto: '🌓 自動', light: '☀️ 淺色', dark: '🌙 深色' };
      document.getElementById('theme-mode-btn').textContent = btnMap[themeMode];
    }

    function cycleThemeMode() {
      const order = ['auto', 'light', 'dark'];
      themeMode = order[(order.indexOf(themeMode) + 1) % order.length];
      localStorage.setItem(THEME_MODE_KEY, themeMode);
      applyDayNightTheme(new Date());
    }

    let isSleeping = false;
    let manualSleepForced = false;
    let wakeOverrideUntil = 0;
    let hintTimeout = null;

    function isNightHours(dateObj) {
      const hour = dateObj.getHours();
      return hour >= 0 && hour < 6;
    }

    function enterSleepMode() {
      if (isSleeping) return;
      isSleeping = true;
      const overlay = document.getElementById('night-overlay');
      const hint = document.getElementById('night-hint');
      overlay.classList.remove('hidden');
      hint.classList.remove('opacity-0');
      hint.classList.add('opacity-100');

      if (hintTimeout) clearTimeout(hintTimeout);
      hintTimeout = setTimeout(() => {
        hint.classList.remove('opacity-100');
        hint.classList.add('opacity-0');
      }, 3000);
    }

    function exitSleepMode() {
      if (!isSleeping) return;
      isSleeping = false;
      document.getElementById('night-overlay').classList.add('hidden');
      if (hintTimeout) clearTimeout(hintTimeout);
      refreshAll(true);
    }

    function wakeUpTemporarily() {
      manualSleepForced = false;
      wakeOverrideUntil = Date.now() + 60 * 1000;
      exitSleepMode();
      checkSleepState(new Date());
    }

    function activateSleepNow() {
      wakeOverrideUntil = 0;
      manualSleepForced = true;
      enterSleepMode();
      document.getElementById('wake-badge').classList.add('hidden');
    }

    function checkSleepState(now) {
      const inNightWindow = isNightHours(now);
      const wakeBadge = document.getElementById('wake-badge');
      const wakeRemEl = document.getElementById('wake-remaining');

      if (!inNightWindow && now.getHours() === 6 && now.getMinutes() === 0 && now.getSeconds() === 0) {
        manualSleepForced = false;
      }

      if (manualSleepForced) {
        wakeBadge.classList.add('hidden');
        enterSleepMode();
        return;
      }

      if (inNightWindow) {
        const remainingWakeSec = Math.ceil((wakeOverrideUntil - Date.now()) / 1000);
        if (remainingWakeSec > 0) {
          wakeBadge.classList.remove('hidden');
          wakeRemEl.textContent = remainingWakeSec;
          if (isSleeping) exitSleepMode();
        } else {
          wakeBadge.classList.add('hidden');
          enterSleepMode();
        }
      } else {
        wakeBadge.classList.add('hidden');
        if (isSleeping) exitSleepMode();
      }
    }

    function updateClock() {
      const now = new Date();
      const hh = String(now.getHours()).padStart(2, '0');
      const mm = String(now.getMinutes()).padStart(2, '0');
      const ss = String(now.getSeconds()).padStart(2, '0');

      flipClockCard('clock-hh', hh);
      flipClockCard('clock-mm', mm);
      flipClockCard('clock-ss', ss);

        month: 'short', day: 'numeric', weekday: 'short'
      });

      checkSleepState(now);
      applyDayNightTheme(now);

      if (!isSleeping && !isFetchingEtas) {
        refreshCountdown--;
        if (refreshCountdown <= 0) {
          refreshCountdown = REFRESH_INTERVAL_SEC;
          refreshAll(false);
        }
        document.getElementById('countdown').textContent = refreshCountdown;
      }
    }

    function renderEtaFlapBoard(containerId, etas, emptyMessage = '暫無班次預報') {

          ? Math.max(0, item.diffMin)
          : Math.max(0, Math.floor(diffMs / 60000));

        const timeStr = etaDate.toLocaleTimeString('zh-HK', { hour: '2-digit', minute: '2-digit', hour12: false });

          `;
        } else {
          const paddedMin = String(diffMin).padStart(2, '0');
          countdownFlap = `
            <div class="flex items-center gap-1">
              <div class="flap-tile ${animClass} px-2.5 py-0.5 text-2xl md:text-3xl font-extrabold mono-num ${idx === 0 ? 'flap-amber' : ''}">
                ${paddedMin}
              </div>
              <span class="text-xs md:text-sm font-black text-sec">分鐘</span>
            </div>
          `;
        }

            <div class="flex items-center gap-2 min-w-0">

    // 使用單例 Promise 鎖定，確保全頁無論多少張九巴卡都只會下載一次站點清單
    let kmbStopNameMap = null;
    let kmbAllRoutesList = null;
    let kmbBaseDataPromise = null;
    let gmbChungUkTargets = null;
    let gmbChungUkPromise = null;

    async function ensureKmbBaseData() {
      if (kmbStopNameMap && kmbAllRoutesList) return;
      if (kmbBaseDataPromise) return kmbBaseDataPromise;

      kmbBaseDataPromise = (async () => {
        try {
          const [stopsRes, routesRes] = await Promise.all([
            fetchJsonWithTimeout('https://data.etabus.gov.hk/v1/transport/kmb/stop', 10000),
            fetchJsonWithTimeout('https://data.etabus.gov.hk/v1/transport/kmb/route/', 8000)
          ]);
          kmbStopNameMap = new Map((stopsRes.data || []).map(s => [s.stop, s]));
          kmbAllRoutesList = routesRes.data || [];
        } catch (err) {
          kmbBaseDataPromise = null;
          throw err;
        }
      })();

      return kmbBaseDataPromise;
    }

    async function hydrateDefaultKmbCard(card) {
      if (card.operator !== 'KMB') return;
      if (card.stopId && card.fiveStopSeqs && card.fiveStopSeqs.length === 5) return;

      await ensureKmbBaseData();

      const isChungUk = (name) => name && (name.includes('鍾屋村') || name.includes('鐘屋村'));
      const [inRes, outRes] = await Promise.all([
        fetchJsonWithTimeout(`https://data.etabus.gov.hk/v1/transport/kmb/route-stop/${card.route}/inbound/${card.serviceType || '1'}`, 6000).catch(() => ({ data: [] })),
        fetchJsonWithTimeout(`https://data.etabus.gov.hk/v1/transport/kmb/route-stop/${card.route}/outbound/${card.serviceType || '1'}`, 6000).catch(() => ({ data: [] }))
      ]);

      const inMeta = kmbAllRoutesList.find(r => r.route === card.route && r.bound === 'I' && String(r.service_type) === String(card.serviceType || '1'));
      const outMeta = kmbAllRoutesList.find(r => r.route === card.route && r.bound === 'O' && String(r.service_type) === String(card.serviceType || '1'));

      let chosenList = [];
      let chosenBound = card.bound;

      if (card.destName.includes('大埔') && inMeta?.dest_tc?.includes('大埔')) {
        chosenList = inRes.data || [];
        chosenBound = 'I';
      } else if (card.destName.includes('大埔') && outMeta?.dest_tc?.includes('大埔')) {
        chosenList = outRes.data || [];
        chosenBound = 'O';
      } else if (card.destName.includes('元朗') && inMeta?.dest_tc?.includes('元朗')) {
        chosenList = inRes.data || [];
        chosenBound = 'I';
      } else if (card.destName.includes('元朗') && outMeta?.dest_tc?.includes('元朗')) {
        chosenList = outRes.data || [];
        chosenBound = 'O';
      } else if ((inRes.data || []).length > 0) {
        chosenList = inRes.data;
        chosenBound = 'I';
      } else {
        chosenList = outRes.data || [];
        chosenBound = 'O';
      }

      const targetIdx = chosenList.findIndex(item => isChungUk(kmbStopNameMap.get(item.stop)?.name_tc));
      if (targetIdx !== -1) {
        const targetItem = chosenList[targetIdx];
        card.bound = chosenBound;
        card.stopId = targetItem.stop;
        card.stopSeq = Number(targetItem.seq);
        card.stopName = kmbStopNameMap.get(targetItem.stop)?.name_tc || card.stopName;

        // 若是 64K 往大埔方向，一併記錄太和廣場站 ID 供交通預測計算實時車程
        if (card.route === '64K' && card.destName.includes('大埔')) {
          const twHit = chosenList.find(item => {
            const nm = kmbStopNameMap.get(item.stop)?.name_tc || '';
            return nm.includes('太和廣場') || nm.includes('太和');
          });
          if (twHit) card.taiWoStopId = twHit.stop;
        }

        const startIdx = Math.max(0, targetIdx - 4);
        const slice = chosenList.slice(startIdx, targetIdx + 1);
        while (slice.length < 5 && slice.length > 0) slice.unshift(slice[0]);

        card.fiveStops = slice.map(s => kmbStopNameMap.get(s.stop)?.name_tc || '車站');
        card.fiveStopIds = slice.map(s => s.stop);
        card.fiveStopSeqs = slice.map(s => Number(s.seq));

        // 立即儲存已解析的 stopId 至 localStorage，下次打開無需再重新搜尋
        saveActiveCards();
        const stopLabelEl = document.getElementById(`card-stop-${ activeCards.indexOf(card) }`);
        if (stopLabelEl) stopLabelEl.textContent = `本站：${card.stopName}`;
      }
    }

    async function fetchKmbCardData(slotIdx, card) {
      try {
        await hydrateDefaultKmbCard(card);

        if (!card.stopId) {
          renderEtaFlapBoard(`eta-card-${slotIdx}`, [], '正在定位巴士站點...');
          return;
        }

        // 1. 直接使用最穩定的九巴單站 ETA API (/eta/{stop_id}/{route}/1) 獲取到站時間
        const etaJson = await fetchJsonWithTimeout(`https://data.etabus.gov.hk/v1/transport/kmb/eta/${card.stopId}/${card.route}/${card.serviceType || '1'}`, 4500);
        const rawList = (etaJson.data || []).filter(e => Boolean(e.eta));

        // 優先按方向或目的地篩選；若該站只有單一方向則直接採用全部班次
        const cleanDestKey = (card.destName || '').replace(/（.*）|\(.*\)|站|總站/g, '').trim().slice(0, 2);
        let filteredRaw = rawList.filter(e => {
          if (cleanDestKey && e.dest_tc && e.dest_tc.includes(cleanDestKey)) return true;
          if (card.bound && e.dir === card.bound) return true;
          return false;
        });
        if (filteredRaw.length === 0) filteredRaw = rawList;

        filteredRaw.sort((a, b) => new Date(a.eta) - new Date(b.eta));

        const formattedEtas = filteredRaw.map(e => ({
          timestamp: e.eta,
          remark: e.rmk_tc || '實時班次'
        }));

        // 2. 若是 64K 往大埔方向且有太和站 ID，非阻塞計算鍾屋村->太和實時車程差
        if (card.route === '64K' && card.destName.includes('大埔') && card.taiWoStopId && filteredRaw.length > 0) {
          latestKmbRemarks = filteredRaw[0]?.rmk_tc || '';
          fetchJsonWithTimeout(`https://data.etabus.gov.hk/v1/transport/kmb/eta/${card.taiWoStopId}/64K/1`, 3500)
            .then(twJson => {
              const twList = (twJson.data || []).filter(e => e.eta && (!e.dest_tc || e.dest_tc.includes('大埔')));
              const t1 = new Date(filteredRaw[0].eta).getTime();
              for (const tw of twList) {
                const diffMin = Math.round((new Date(tw.eta).getTime() - t1) / 60000);
                if (diffMin >= 5 && diffMin <= 32) {
                  latestCorridorMinutes = diffMin;
                  evaluateAndRenderAllTrafficMonitors();
                  break;
                }
              }
            })
            .catch(() => {});
        }

        renderEtaFlapBoard(`eta-card-${slotIdx}`, formattedEtas, '目前暫無班次預報');
        updateJrApproachBoard(slotIdx, card, formattedEtas, null);
      } catch (err) {
        console.error('KMB ETA Error:', err);
        renderEtaFlapBoard(`eta-card-${slotIdx}`, [], '暫時無法連線九巴伺服器，5秒後重試');
      }
    }

    async function initGmb25KChungUkTargets(card) {

                });

            }
          }
          if (matchedTargets.length > 0) {
            gmbChungUkTargets = matchedTargets;
            return gmbChungUkTargets;
          }
        } catch (e) {}

        gmbChungUkPromise = null;

    async function fetchGmbCardData(slotIdx, card) {
      try {
        const targets = card.isDefaultChungUk25K
          ? await initGmb25KChungUkTargets(card)
          : [{ route_id: card.routeId, route_seq: card.routeSeq, stop_seq: card.stopSeq, stop_id: card.stopId, name_tc: card.stopName, dest_tc: card.destName }];

        const collectedEtas = [];
        const routeStopPromises = targets.map(t =>
          fetchJsonWithTimeout(`https://data.etagmb.gov.hk/eta/route-stop/${t.route_id}/${t.route_seq}/${t.stop_seq}`, 4500)
            .then(json => ({ json, target: t })).catch(() => null)
        );
        const uniqueStopIds = [...new Set(targets.map(t => t.stop_id).filter(Boolean))];
        const stopPromises = uniqueStopIds.map(stopId =>
          fetchJsonWithTimeout(`https://data.etagmb.gov.hk/eta/stop/${stopId}`, 4500).catch(() => null)
        );

        const [routeStopResults, stopResults] = await Promise.all([Promise.all(routeStopPromises), Promise.all(stopPromises)]);

        for (const item of routeStopResults) {
          if (!item || !item.json) continue;
          const data = item.json.data;
          let etaList = [];
          if (Array.isArray(data)) data.forEach(d => { if (Array.isArray(d.eta)) etaList.push(...d.eta); });
          else if (data && Array.isArray(data.eta)) etaList = data.eta;

          for (const e of etaList) {
            const ts = e.timestamp || (e.diff !== undefined ? new Date(Date.now() + e.diff * 60000).toISOString() : null);
            if (ts) collectedEtas.push({ timestamp: ts, diffMin: e.diff, remark: e.remarks_tc || `往${item.target.dest_tc}` });
          }
        }

        if (collectedEtas.length === 0) {
          const validRouteSeqKeys = new Set(targets.map(t => `${t.route_id}_${t.route_seq}`));
          for (const sJson of stopResults) {
            const arr = sJson?.data;
            if (!Array.isArray(arr)) continue;
            for (const entry of arr) {
              if (validRouteSeqKeys.size === 0 || validRouteSeqKeys.has(`${entry.route_id}_${entry.route_seq}`)) {
                for (const e of (entry.eta || [])) {
                  const ts = e.timestamp || (e.diff !== undefined ? new Date(Date.now() + e.diff * 60000).toISOString() : null);
                  if (ts) collectedEtas.push({ timestamp: ts, diffMin: e.diff, remark: e.remarks_tc || `往${card.destName}` });
                }
              }
            }
          }
        }

        const seenMinutes = new Set();
        const uniqueSortedEtas = collectedEtas
          .sort((a, b) => new Date(a.timestamp) - new Date(b.timestamp))
          .filter(item => {
            const minuteBucket = Math.floor(new Date(item.timestamp).getTime() / 30000);
            if (seenMinutes.has(minuteBucket)) return false;
            seenMinutes.add(minuteBucket);
            return true;
          });

        renderEtaFlapBoard(`eta-card-${slotIdx}`, uniqueSortedEtas, `目前暫無 ${card.route} 到站預報`);
        updateJrApproachBoard(slotIdx, card, uniqueSortedEtas, null);
      } catch (err) {
        renderEtaFlapBoard(`eta-card-${slotIdx}`, [], '讀取小巴資料失敗');
      }
    }

    async function fetchCtbCardData(slotIdx, card) {
      try {
        const json = await fetchJsonWithTimeout(`https://rt.data.gov.hk/v2/transport/citybus/eta/ctb/${card.stopId}/${card.route}`, 4500);
        const rawList = (json.data || []).filter(e => e.eta && (!card.bound || e.dir === card.bound));
        const formatted = rawList.map(e => ({ timestamp: e.eta, remark: e.rmk_tc || `往${e.dest_tc || card.destName}` }));

        renderEtaFlapBoard(`eta-card-${slotIdx}`, formatted, `暫無城巴 ${card.route} 班次`);
        updateJrApproachBoard(slotIdx, card, formatted, null);
      } catch (err) {
        renderEtaFlapBoard(`eta-card-${slotIdx}`, [], '讀取城巴資料失敗');
      }
    }

    async function fetchMtrCardData(slotIdx, card) {
      try {
        const json = await fetchJsonWithTimeout(`https://rt.data.gov.hk/v1/transport/mtr/getSchedule.php?line=${card.route}&sta=${card.stopId}&lang=TC`, 4500);
        const trains = json?.data?.[`${card.route}-${card.stopId}`]?.[card.bound] || [];
        const formatted = trains.map(t => {
          const min = Number(t.ttnt) || 0;
          return { timestamp: new Date(Date.now() + min * 60000).toISOString(), diffMin: min, remark: `${t.plat ? t.plat + '號月台' : '實時列車'}` };
        });

        renderEtaFlapBoard(`eta-card-${slotIdx}`, formatted, '目前暫無列車班次');
        updateJrApproachBoard(slotIdx, card, formatted, null);
      } catch (err) {
        renderEtaFlapBoard(`eta-card-${slotIdx}`, [], '讀取港鐵資料失敗');
      }
    }

    async function onModalRouteSelected() {
      const routeCode = document.getElementById('modal-route-select').value;
      const dirSelect = document.getElementById('modal-dir-select');
      if (!routeCode) return;

      dirSelect.innerHTML = '<option value="">載入方向中...</option>';

      if (modalOperator === 'KMB') {
        await ensureKmbBaseData();
        const matches = kmbAllRoutesList.filter(r => r.route === routeCode && String(r.service_type) === '1');
        modalDirectionsPool = matches.map(m => ({ key: `${m.bound}_${m.service_type}`, bound: m.bound, serviceType: m.service_type, destName: m.dest_tc, label: `往 ${m.dest_tc}（由 ${m.orig_tc} 開出）` }));
      } else if (modalOperator === 'GMB') {
        const json = await fetchJsonWithTimeout(`https://data.etagmb.gov.hk/route/${modalGmbRegion}/${routeCode}`, 5000);
        modalDirectionsPool = [];
        for (const v of (json.data || [])) {
          const desc = v.description_tc || '常規班次';
          for (const d of (v.directions || [])) {
            modalDirectionsPool.push({ key: `${v.route_id}_${d.route_seq}`, routeId: v.route_id, routeSeq: d.route_seq, destName: d.dest_tc, label: `往 ${d.dest_tc}（${desc}）` });
          }
        }
      } else if (modalOperator === 'CTB') {
        const routeMeta = modalRoutesPool.find(r => r.code === routeCode);
        modalDirectionsPool = [
          { key: 'I', bound: 'I', destName: routeMeta?.orig_tc || '返程', label: `往 ${routeMeta?.orig_tc || 'Inbound'}` },
          { key: 'O', bound: 'O', destName: routeMeta?.dest_tc || '去程', label: `往 ${routeMeta?.dest_tc || 'Outbound'}` }
        ];
      } else if (modalOperator === 'MTR') {
        const lineInfo = MTR_LINES_DATA[routeCode];
        const firstSta = lineInfo.stations[0].name;
        const lastSta = lineInfo.stations[lineInfo.stations.length - 1].name;
        modalDirectionsPool = [
          { key: 'UP', bound: 'UP', destName: lastSta, label: `往 ${lastSta} 方向 (UP)` },
          { key: 'DOWN', bound: 'DOWN', destName: firstSta, label: `往 ${firstSta} 方向 (DOWN)` }
        ];
      }

      dirSelect.innerHTML = modalDirectionsPool.map((d, i) => `<option value="${i}">${d.label}</option>`).join('');
      onModalDirSelected();
    }

    async function onModalDirSelected() {
      const routeCode = document.getElementById('modal-route-select').value;
      const dirIdx = document.getElementById('modal-dir-select').value;
      const dirObj = modalDirectionsPool[dirIdx];
      const stopSelect = document.getElementById('modal-stop-select');
      if (!dirObj) return;

      stopSelect.innerHTML = '<option value="">載入沿線站點中...</option>';
      modalStopsPool = [];

      if (modalOperator === 'KMB') {
        await ensureKmbBaseData();
        const boundWord = dirObj.bound === 'I' ? 'inbound' : 'outbound';
        const json = await fetchJsonWithTimeout(`https://data.etabus.gov.hk/v1/transport/kmb/route-stop/${routeCode}/${boundWord}/${dirObj.serviceType}`, 5000);
        modalStopsPool = (json.data || []).map(s => ({ stopId: s.stop, stopSeq: Number(s.seq), name_tc: kmbStopNameMap.get(s.stop)?.name_tc || s.stop }));
      } else if (modalOperator === 'GMB') {
        const json = await fetchJsonWithTimeout(`https://data.etagmb.gov.hk/route-stop/${dirObj.routeId}/${dirObj.routeSeq}`, 5000);
        modalStopsPool = (json?.data?.route_stops || []).map(s => ({ stopId: s.stop_id, stopSeq: Number(s.stop_seq), name_tc: s.name_tc }));
      } else if (modalOperator === 'CTB') {
        const boundWord = dirObj.bound === 'I' ? 'inbound' : 'outbound';
        const json = await fetchJsonWithTimeout(`https://rt.data.gov.hk/v2/transport/citybus/route-stop/ctb/${routeCode}/${boundWord}`, 5000);
        const rawStops = json.data || [];
        modalStopsPool = await Promise.all(
          rawStops.map(s =>
            fetchJsonWithTimeout(`https://rt.data.gov.hk/v2/transport/citybus/stop/${s.stop}`, 4000)
              .then(j => ({ stopId: s.stop, stopSeq: Number(s.seq), name_tc: j?.data?.name_tc || s.stop }))
              .catch(() => ({ stopId: s.stop, stopSeq: Number(s.seq), name_tc: s.stop }))
          )
        );
      } else if (modalOperator === 'MTR') {
        const stations = [...MTR_LINES_DATA[routeCode].stations];
        if (dirObj.bound === 'DOWN') stations.reverse();
        modalStopsPool = stations.map((st, idx) => ({ stopId: st.id, stopSeq: idx + 1, name_tc: `${st.name}站` }));
      }

      stopSelect.innerHTML = modalStopsPool.map((s, idx) => `<option value="${idx}">${idx + 1}. ${s.name_tc}</option>`).join('');
    }

    function saveModalRouteCard() {
      const routeCode = document.getElementById('modal-route-select').value;
      const dirIdx = document.getElementById('modal-dir-select').value;
      const stopIdx = Number(document.getElementById('modal-stop-select').value);

      const dirObj = modalDirectionsPool[dirIdx];
      const targetStop = modalStopsPool[stopIdx];
      if (!routeCode || !dirObj || !targetStop) return;

      const startIdx = Math.max(0, stopIdx - 4);
      const slice = modalStopsPool.slice(startIdx, stopIdx + 1);
      while (slice.length < 5 && slice.length > 0) slice.unshift(slice[0]);

      const newCard = {
        operator: modalOperator, gmbRegion: modalGmbRegion, route: routeCode,
        bound: dirObj.bound || 'I', serviceType: dirObj.serviceType || '1',
        routeId: dirObj.routeId || null, routeSeq: dirObj.routeSeq || null,
        destName: dirObj.destName, stopId: targetStop.stopId, stopSeq: targetStop.stopSeq,
        stopName: targetStop.name_tc,
        fiveStops: slice.map(s => s.name_tc), fiveStopIds: slice.map(s => s.stopId), fiveStopSeqs: slice.map(s => Number(s.stopSeq)),
        isDefaultChungUk25K: false
      };

      if (isAddingNewRoute) activeCards.push(newCard);
      else activeCards[modalEditingSlot] = newCard;

      saveActiveCards();
      closeRouteModal();
      buildCardsSkeleton();
      refreshAll(true);
    }

      // 背景更新天氣與交通消息 (不阻塞 5 秒巴士到站更新)

        fetchTdTrafficNews().catch(() => {});
      }
      if (isManualOrInitial || slowSyncCounter % 12 === 0) {
        fetchHkoSunTimes().catch(() => {});
        fetchWeather().catch(() => {});
      }
      slowSyncCounter++;

      if (isFetchingEtas && !isManualOrInitial) return;
      isFetchingEtas = true;

      try {
        await fetchAllActiveCards();
        const nowStr = new Date().toLocaleTimeString('zh-HK', { hour12: false });
        document.getElementById('last-updated').textContent = `最後更新：${nowStr}`;
      } finally {
        isFetchingEtas = false;
      }
    }
